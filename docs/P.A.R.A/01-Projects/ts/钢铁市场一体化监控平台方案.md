```python
# 预警系统核心代码
class AlertSystem:
    def __init__(self):
        self.alert_rules = {
            'red_alert': {
                'conditions': [
                    'pmi连续3个月低于45',
                    '房地产投资增速下降>15%',
                    '钢材库存连续6周上升>20%'
                ],
                'level': '高危',
                'action': '立即通知'
            },
            'orange_alert': {
                'conditions': [
                    'pmi连续2个月低于50',
                    '价格波动率超过5%',
                    '资金连续流出>5天'
                ],
                'level': '中危',
                'action': '邮件通知'
            },
            'yellow_alert': {
                'conditions': [
                    '重要数据发布前1-2天',
                    '政策会议期间',
                    '节假日前后'
                ],
                'level': '提醒',
                'action': '系统提示'
            }
        }

    def check_alerts(self):
        alerts = []
        for rule_name, rule in self.alert_rules.items():
            if self.evaluate_conditions(rule['conditions']):
                alerts.append({
                    'rule': rule_name,
                    'level': rule['level'],
                    'action': rule['action'],
                    'timestamp': datetime.now()
                })
        return alerts
```

# 数据采集自动化
class DataCollector:
    def __init__(self):
        self.collectors = {
            'pmi': self.collect_pmi_data,
            'bond': self.collect_bond_data,
            'future': self.collect_future_data,
            'inventory': self.collect_inventory_data,
            'sentiment': self.collect_sentiment_data
        }

    def collect_all_data(self):
        data = {}
        for name, collector in self.collectors.items():
            try:
                data[name] = collector()
                logger.info(f"成功采集{name}数据")
            except Exception as e:
                logger.error(f"采集{name}数据失败: {e}")
        return data

    def schedule_collection(self):
        # 设置定时采集任务
        schedule.every().day.at("09:00").do(self.collect_all_data)
        schedule.every().wednesday.at("15:00").do(self.collect_inventory_data)
        schedule.every().monday.at("10:00").do(self.collect_pmi_data)

        while True:
            schedule.run_pending()
            time.sleep(60)
